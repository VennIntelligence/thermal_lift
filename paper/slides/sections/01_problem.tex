\section{Problem and Claim Boundary}

\begin{frame}{Inspection Goal}
  \begin{itemize}
    \item Industrial chip inspection needs clearer internal thermal contours.
    \item Input data are raw LWIR temperature matrices, not rendered color video.
    \item The goal is contour visibility on a 2x grid, not absolute temperature SR.
  \end{itemize}
\end{frame}

\begin{frame}{What We Can and Cannot Claim}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \textbf{Can claim}
      \begin{itemize}
        \item 255-frame main-session POC.
        \item Data-driven alignment improves registration evidence.
        \item 2x contour-level reconstruction is visually and quantitatively gated.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Cannot claim}
      \begin{itemize}
        \item 4x SR.
        \item 5 um actual spatial resolution.
        \item Absolute-temperature metrology.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
