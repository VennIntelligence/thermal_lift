% ---------------------------------------------------------------- 原理：零空间命题
\begin{frame}{原理：为什么观测侧锚定对漂移“结构性失明”}
  \small 任意 HR 扰动分解为可观测分量与盲区分量：
  \[
    \delta x = \delta x_{\text{range}} + \delta x_{\text{null}},
    \qquad A\,\delta x_{\text{null}} = 0 .
  \]
  \begin{block}{命题 P1（精确零空间不变性）}
    对任意只经 $A$ 接触预测的观测域损失 $\mathcal{L}(\hat{x})=\ell(A\hat{x},y)$：
    \[
      \mathcal{L}(\hat{x}+\delta x_{\text{null}})=\mathcal{L}(\hat{x}),
      \qquad
      \langle \nabla\mathcal{L}(\hat{x}),\,\delta x_{\text{null}}\rangle
      =\langle \nabla_1\ell,\,A\,\delta x_{\text{null}}\rangle = 0 .
    \]
  \end{block}
  \footnotesize $\epsilon$-零空间（P2/P3）：在近零奇异方向 $\sigma_i\!\to\!0$ 上，观测损失灵敏度 $O(\epsilon)$、
  平方损失恢复曲率 $\|WAv_i\|^2\!\le\!\|W\|^2\sigma_i^2 \to 0$。
  \takeaway{\redflag{任何有界观测侧锚定（有限权重、任意有界频带）对 $\delta x_{\text{null}}$ 都恒盲}——这是定理，不是调参问题。}
\end{frame}
